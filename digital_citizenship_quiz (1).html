⚙️ Google Sheet 연동 설정

URL을 입력하고 저장하면 학생 제출 응답이 해당 구글 시트에 자동 기록됩니다. 입력하지 않으면 시뮬레이션 모드로 작동합니다.

📖 구글 시트 연결 방법 (Apps Script)

  1. 새 구글 시트 생성 ➔ [확장 프로그램] ➔ [Apps Script] 클릭
  2. 아래 코드 복사 후 기존 코드를 대체하여 붙여넣기
  3. 우측 상단 [배포] ➔ [새 배포] 클릭
  4. 유형 선택: 웹 앱 / 액세스 권한: 모든 사용자(Anyone) 설정 후 배포
  5. 발급된 웹 앱 URL을 위 입력창에 붙여넣기 후 저장!
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // 헤더가 없으면 생성
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['제출시각', '출석번호', '문항1', '문항2', '문항3', '문항4', '문항5', '점수']);
    }
    
    // 응답 저장
    sheet.appendRow([
      data.timestamp,
      data.studentNum,
      data.answers['Q1'] || '',
      data.answers['Q2'] || '',
      data.answers['Q3'] || '',
      data.answers['Q4'] || '',
      data.answers['Q5'] || '',
      data.score
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
